**Créer lib/services/search_history_service.dart:**

```dart
import 'package:shared_preferences/shared_preferences.dart';

/// Service de gestion de l'historique de recherche.
/// Stocke les requêtes dans SharedPreferences (clé 'search_history').
/// Max 10 entrées, LIFO, sans doublons.
class SearchHistoryService {
  static const String _historyKey = 'search_history';
  static const int _maxHistory = 10;

  /// Retourne la liste des recherches récentes (ordre LIFO — la plus récente en index 0)
  static Future<List<String>> getHistory() async {
    final prefs = await SharedPreferences.getInstance();
    return prefs.getStringList(_historyKey) ?? [];
  }

  /// Ajoute une query à l'historique.
  /// - Déduplique : si la query existe déjà, elle est retirée puis réinsérée en tête.
  /// - Limite : si la liste dépasse 10 entrées après insertion, la dernière est supprimée.
  static Future<void> addToHistory(String query) async {
    final trimmed = query.trim();
    if (trimmed.isEmpty) return;
    final prefs = await SharedPreferences.getInstance();
    final history = prefs.getStringList(_historyKey) ?? [];
    history.remove(trimmed); // déduplique
    history.insert(0, trimmed); // LIFO
    if (history.length > _maxHistory) history.removeLast();
    await prefs.setStringList(_historyKey, history);
  }

  /// Supprime toutes les entrées de l'historique.
  static Future<void> clearHistory() async {
    final prefs = await SharedPreferences.getInstance();
    await prefs.remove(_historyKey);
  }
}
```

Future<void> _fetchSuggestions(String prefix) async {
  final sanitized = prefix.trim().replaceAll('"', '');
  if (sanitized.length < 2) return;
  setState(() => _isFetchingSuggestions = true);
  try {
    final where = Uri.encodeQueryComponent('search(marque_produit,"$sanitized")');
    final url = '${AppConstants.apiBaseUrl}/records'
        '?where=$where&select=marque_produit&group_by=marque_produit&limit=8';
    final response = await http.get(Uri.parse(url)).timeout(const Duration(seconds: 5));
    if (!mounted) return;
    if (response.statusCode == 200) {
      final data = jsonDecode(response.body);
      final results = (data['results'] as List? ?? [])
          .map((r) => (r['marque_produit'] ?? '').toString())
          .where((s) => s.isNotEmpty)
          .toList();
      setState(() {
        _suggestions = results;
        _isFetchingSuggestions = false;
      });
    } else {
      setState(() { _suggestions = []; _isFetchingSuggestions = false; });
    }
  } catch (_) {
    if (mounted) setState(() { _suggestions = []; _isFetchingSuggestions = false; });
  }
}
