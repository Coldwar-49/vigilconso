**Créer lib/services/read_status_service.dart:**

```dart
import 'package:shared_preferences/shared_preferences.dart';

/// Service de gestion du statut lu/non-lu des rappels.
/// Stocke les IDs des rappels lus dans SharedPreferences (clé 'read_rappels').
/// ID unique: reference_fiche ?? libelle ?? numero_fiche
class ReadStatusService {
  static const String _readKey = 'read_rappels';

  /// Retourne l'ID unique d'un rappel (même logique que FavoritesService)
  static String getRappelId(dynamic rappel) {
    return rappel['reference_fiche']?.toString()
        ?? rappel['libelle']?.toString()
        ?? rappel['numero_fiche']?.toString()
        ?? '';
  }

  /// Charge l'ensemble des IDs lus depuis SharedPreferences.
  static Future<Set<String>> getReadIds() async {
    final prefs = await SharedPreferences.getInstance();
    final raw = prefs.getStringList(_readKey) ?? [];
    return Set<String>.from(raw);
  }

  /// Marque un rappel comme lu.
  /// [id] : l'ID unique du rappel (reference_fiche ?? libelle ?? numero_fiche)
  static Future<void> markAsRead(String id) async {
    if (id.isEmpty) return;
    final prefs = await SharedPreferences.getInstance();
    final raw = prefs.getStringList(_readKey) ?? [];
    final readSet = Set<String>.from(raw);
    if (readSet.contains(id)) return; // déjà lu — évite une écriture inutile
    readSet.add(id);
    await prefs.setStringList(_readKey, readSet.toList());
  }

  /// Retourne true si le rappel avec cet ID a déjà été lu.
  static Future<bool> isRead(String id) async {
    if (id.isEmpty) return false;
    final readSet = await getReadIds();
    return readSet.contains(id);
  }

  /// Retourne le nombre de rappels non-lus parmi une liste de rappels.
  /// Utilise les IDs déjà chargés pour éviter plusieurs accès SharedPreferences.
  static Future<int> getUnreadCount(List<dynamic> rappels) async {
    if (rappels.isEmpty) return 0;
    final readSet = await getReadIds();
    int unread = 0;
    for (final r in rappels) {
      final id = getRappelId(r);
      if (id.isNotEmpty && !readSet.contains(id)) unread++;
    }
    return unread;
  }
}
```

// ignore: depend_on_referenced_packages
import 'package:flutter_test/flutter_test.dart';
import 'package:shared_preferences/shared_preferences.dart';

void main() {

    test('isRead — retourne false initialement pour tout rappel', () async {
      final result = await ReadStatusService.isRead('REF-001');
      expect(result, isFalse);
    });

    test('markAsRead — persiste dans SharedPreferences', () async {
      await ReadStatusService.markAsRead('REF-001');
      final result = await ReadStatusService.isRead('REF-001');
      expect(result, isTrue);
    });

    test('markAsRead — idempotent (appels multiples sans doublon)', () async {
      await ReadStatusService.markAsRead('REF-001');
      await ReadStatusService.markAsRead('REF-001');
      final readSet = await ReadStatusService.getReadIds();
      expect(readSet.where((id) => id == 'REF-001').length, equals(1));
    });

    test('markAsRead — ignore les IDs vides', () async {
      await ReadStatusService.markAsRead('');
      final readSet = await ReadStatusService.getReadIds();
      expect(readSet, isEmpty);
    });

    test('getReadIds — retourne tous les IDs marqués lus', () async {
      await ReadStatusService.markAsRead('REF-001');
      await ReadStatusService.markAsRead('REF-002');
      final readSet = await ReadStatusService.getReadIds();
      expect(readSet.contains('REF-001'), isTrue);
      expect(readSet.contains('REF-002'), isTrue);
      expect(readSet.length, equals(2));
    });

    test('getUnreadCount — compte les non-lus dans une liste', () async {
      await ReadStatusService.markAsRead('REF-001');
      final rappels = [
        {'reference_fiche': 'REF-001'},
        {'reference_fiche': 'REF-002'},
        {'reference_fiche': 'REF-003'},
      ];
      final count = await ReadStatusService.getUnreadCount(rappels);
      expect(count, equals(2));
    });

    test('getUnreadCount — liste vide retourne 0', () async {
      final count = await ReadStatusService.getUnreadCount([]);
      expect(count, equals(0));
    });

    test('getRappelId — utilise reference_fiche en priorité', () {
      final rappel = {'reference_fiche': 'REF-001', 'libelle': 'Nutella', 'numero_fiche': 'NUM-001'};
