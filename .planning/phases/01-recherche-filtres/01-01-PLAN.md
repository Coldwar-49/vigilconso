Future<void> _performSearch(String query) async {
  final q = query.trim().replaceAll('"', ''); // sanitize avant injection URL
  if (q.isEmpty) return;
  setState(() { _isLoading = true; _errorMessage = null; _hasSearched = true; });
  try {
    final result = await RappelService.fetchRappels(
      categoryKey: 'all',
      searchQuery: q,
      newestFirst: _isNewestFirst,
    );
    if (!mounted) return;
    if (result.containsKey('error')) {
      setState(() { _isLoading = false; _errorMessage = result['error']; });
      return;
    }
    final results = List<dynamic>.from(result['results'] ?? []);
    setState(() {
      _allResults = results;
      _filteredResults = List.from(results);
      _isLoading = false;
    });
    _staggerController.forward(from: 0);
  } catch (e) {
    if (!mounted) return;
    setState(() { _isLoading = false; _errorMessage = 'Impossible de charger les résultats. Vérifiez votre connexion et réessayez.'; });
  }
}

        ]),
      ]),

    Expanded(

void _toggleSort() {
  setState(() {
    _isNewestFirst = !_isNewestFirst;
    _filteredResults.sort((a, b) {
      final da = RappelService.parseDate(a['date_publication'] ?? '') ?? DateTime(1900);
      final db = RappelService.parseDate(b['date_publication'] ?? '') ?? DateTime(1900);
      return _isNewestFirst ? db.compareTo(da) : da.compareTo(db);
    });
  });
}
