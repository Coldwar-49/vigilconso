/// Value object représentant l'état courant des filtres.
/// Immuable — copyWith pour créer des variantes modifiées.
class FilterState {
  final Set<String> riskLevels;  // 'grave', 'moderé', 'faible' — vide = tous
  final int? dateDays;           // 7, 30, 90 — null = pas de filtre date preset
  final DateTimeRange? customDateRange; // non-null si 'Personnalisé' sélectionné
  final String brandKeyword;     // '' = pas de filtre marque

  const FilterState({
    this.riskLevels = const {},
    this.dateDays = null,
    this.customDateRange = null,
    this.brandKeyword = '',
  });

  bool get hasActiveFilters =>
    riskLevels.isNotEmpty || dateDays != null || customDateRange != null || brandKeyword.isNotEmpty;

  /// Calcule dateFrom ('YYYY-MM-DD') pour l'appel API selon dateDays ou customDateRange
  String? get apiDateFrom {
    if (customDateRange != null) {
      return '${customDateRange!.start.year.toString().padLeft(4, '0')}-'
             '${customDateRange!.start.month.toString().padLeft(2, '0')}-'
             '${customDateRange!.start.day.toString().padLeft(2, '0')}';
    }
    if (dateDays != null) {
      final d = DateTime.now().subtract(Duration(days: dateDays!));
      return '${d.year.toString().padLeft(4, '0')}-'
             '${d.month.toString().padLeft(2, '0')}-'
             '${d.day.toString().padLeft(2, '0')}';
    }
    return null;
  }

  /// Calcule dateTo ('YYYY-MM-DD') pour le filtre personnalisé
  String? get apiDateTo {
    if (customDateRange != null) {
      return '${customDateRange!.end.year.toString().padLeft(4, '0')}-'
             '${customDateRange!.end.month.toString().padLeft(2, '0')}-'
             '${customDateRange!.end.day.toString().padLeft(2, '0')}';
    }
    return null;
  }

  FilterState copyWith({
    Set<String>? riskLevels,
    int? dateDays,
    bool clearDateDays = false,
    DateTimeRange? customDateRange,
    bool clearCustomDateRange = false,
    String? brandKeyword,
  }) {
    return FilterState(
      riskLevels: riskLevels ?? this.riskLevels,
      dateDays: clearDateDays ? null : (dateDays ?? this.dateDays),
      customDateRange: clearCustomDateRange ? null : (customDateRange ?? this.customDateRange),
      brandKeyword: brandKeyword ?? this.brandKeyword,
    );
  }

  static const FilterState empty = FilterState();
}

// Dans lib/screens/search_screen.dart — fonctions statiques de filtrage

/// Filtre client-side par niveau de risque (OR logic)
/// riskLevels: {'grave', 'modéré', 'faible'} — vide = pas de filtre
static List<dynamic> applyRiskFilter(Set<String> riskLevels, List<dynamic> results) {
  if (riskLevels.isEmpty) return results;
  return results.where((r) {
    final risque = (r['risques_encourus'] ?? '').toString().toLowerCase();
    for (final level in riskLevels) {
      if (level == 'grave' && (risque.contains('grave') || risque.contains('sérieux') || risque.contains('serieux') || risque.contains('élevé') || risque.contains('eleve'))) return true;
      if (level == 'modéré' && (risque.contains('modéré') || risque.contains('moderé') || risque.contains('moyen'))) return true;
      if (level == 'faible' && (risque.contains('faible') || risque.contains('bénin') || risque.contains('benin'))) return true;
    }
    return false;
  }).toList();
}

/// Filtre client-side par marque (case-insensitive contains sur marque_produit)
static List<dynamic> applyBrandFilter(String keyword, List<dynamic> results) {
  if (keyword.isEmpty) return results;
  final kw = keyword.toLowerCase();
  return results.where((r) {
    final brand = (r['marque_produit'] ?? r['nom_marque'] ?? '').toString().toLowerCase();
    return brand.contains(kw);
  }).toList();
}

Future<void> _applyFiltersAndSearch() async {
  final query = _searchController.text.trim().replaceAll('"', '');
  if (query.isEmpty && !_activeFilters.hasActiveFilters) return;

  setState(() { _isLoading = true; _errorMessage = null; });
  try {
    final result = await RappelService.fetchRappels(
      categoryKey: 'all',
      searchQuery: query,
      dateFrom: _activeFilters.apiDateFrom,
      dateTo: _activeFilters.apiDateTo,
      newestFirst: _isNewestFirst,
    );
    if (!mounted) return;
    if (result.containsKey('error')) {
      setState(() { _isLoading = false; _errorMessage = result['error']; });
      return;
    }
    var results = List<dynamic>.from(result['results'] ?? []);
    // Filtres client-side
    results = SearchScreen.applyRiskFilter(_activeFilters.riskLevels, results);
    results = SearchScreen.applyBrandFilter(_activeFilters.brandKeyword, results);
    setState(() {
      _allResults = List<dynamic>.from(result['results'] ?? []);
      _filteredResults = results;
      _hasSearched = true;
      _isLoading = false;
    });
    _staggerController.forward(from: 0);
  } catch (e) {
    if (!mounted) return;
    setState(() { _isLoading = false; _errorMessage = 'Impossible de charger les résultats. Vérifiez votre connexion et réessayez.'; });
  }
}

List<Widget> _buildActiveFilterChips() {
  final cs = Theme.of(context).colorScheme;
  final chips = <Widget>[];

  for (final level in _activeFilters.riskLevels) {
    final label = level == 'grave' ? 'Grave' : level == 'modéré' ? 'Modéré' : 'Faible';
    chips.add(_activeChip(label, cs, () {
      setState(() {
        _activeFilters = _activeFilters.copyWith(
          riskLevels: _activeFilters.riskLevels.difference({level}),
        );
        _applyFiltersAndSearch();
      });
    }));
  }

  if (_activeFilters.dateDays != null) {
    chips.add(_activeChip('${_activeFilters.dateDays} jours', cs, () {
      setState(() { _activeFilters = _activeFilters.copyWith(clearDateDays: true); });
      _applyFiltersAndSearch();
    }));
  }

  if (_activeFilters.customDateRange != null) {
    final r = _activeFilters.customDateRange!;
    final label = '${r.start.day.toString().padLeft(2,'0')}/${r.start.month.toString().padLeft(2,'0')}'
                  ' – ${r.end.day.toString().padLeft(2,'0')}/${r.end.month.toString().padLeft(2,'0')}';
    chips.add(_activeChip(label, cs, () {
      setState(() { _activeFilters = _activeFilters.copyWith(clearCustomDateRange: true); });
      _applyFiltersAndSearch();
    }));
  }

  if (_activeFilters.brandKeyword.isNotEmpty) {
    chips.add(_activeChip(_activeFilters.brandKeyword, cs, () {
      setState(() { _activeFilters = _activeFilters.copyWith(brandKeyword: ''); });
      _applyFiltersAndSearch();
    }));
  }

  return chips;
}
