// Immutable value object for user subscription state.

class Subscriptions {
  final List<String> brands;
  final List<String> categories;

  const Subscriptions({
    required this.brands,
    required this.categories,
  });

  Subscriptions copyWith({List<String>? brands, List<String>? categories}) {
    return Subscriptions(
      brands: brands ?? this.brands,
      categories: categories ?? this.categories,
    );
  }

  bool get isEmpty => brands.isEmpty && categories.isEmpty;
}

```dart
// lib/services/subscription_service.dart
// Mirrors FavoritesService (lib/services/favorites_service.dart) pattern.
// All public mutations go through _lock.synchronized().
// _syncTagsUnlocked is called inside the lock but must NOT call _lock.synchronized() again.
// [SECURITY T-3-01]: Brand names trimmed + truncated to 100 chars before OneSignal Tag sync.

import 'dart:convert';
import 'package:onesignal_flutter/onesignal_flutter.dart';
import 'package:shared_preferences/shared_preferences.dart';
import 'package:synchronized/synchronized.dart';

class SubscriptionService {
  static const String _brandsKey = 'subscription_brands';
  static const String _categoriesKey = 'subscription_categories';
  static final _lock = Lock();

  static Future<List<String>> getBrands() async {
    final prefs = await SharedPreferences.getInstance();
    return List<String>.from(
      jsonDecode(prefs.getString(_brandsKey) ?? '[]') as List,
    );
  }

  static Future<List<String>> getCategories() async {
    final prefs = await SharedPreferences.getInstance();
    return List<String>.from(
      jsonDecode(prefs.getString(_categoriesKey) ?? '[]') as List,
    );
  }

  static Future<void> addBrand(String brand) async {
    await _lock.synchronized(() async {
      final prefs = await SharedPreferences.getInstance();
      final brands = List<String>.from(
        jsonDecode(prefs.getString(_brandsKey) ?? '[]') as List,
      );
      // T-3-01: trim whitespace + truncate to 100 chars before storage/tag sync
      final normalized = brand.trim().toLowerCase();
      final sanitized = brand.trim().length > 100
          ? brand.trim().substring(0, 100)
          : brand.trim();
      if (sanitized.isEmpty) return;
      if (!brands.map((b) => b.toLowerCase()).contains(normalized)) {
        brands.add(sanitized);
        await prefs.setString(_brandsKey, jsonEncode(brands));
        await _syncTagsUnlocked(brands, null);
      }
    });
  }

  static Future<void> removeBrand(String brand) async {
    await _lock.synchronized(() async {
      final prefs = await SharedPreferences.getInstance();
      final brands = List<String>.from(
        jsonDecode(prefs.getString(_brandsKey) ?? '[]') as List,
      );
      brands.removeWhere((b) => b.toLowerCase() == brand.trim().toLowerCase());
      await prefs.setString(_brandsKey, jsonEncode(brands));
      await _syncTagsUnlocked(brands, null);
    });
  }

  static Future<void> addCategory(String categoryKey) async {
    await _lock.synchronized(() async {
      final prefs = await SharedPreferences.getInstance();
      final cats = List<String>.from(
        jsonDecode(prefs.getString(_categoriesKey) ?? '[]') as List,
      );
      if (!cats.contains(categoryKey)) {
        cats.add(categoryKey);
        await prefs.setString(_categoriesKey, jsonEncode(cats));
        await _syncTagsUnlocked(null, cats);
      }
    });
  }

  static Future<void> removeCategory(String categoryKey) async {
    await _lock.synchronized(() async {
      final prefs = await SharedPreferences.getInstance();
      final cats = List<String>.from(
        jsonDecode(prefs.getString(_categoriesKey) ?? '[]') as List,
      );
      cats.remove(categoryKey);
      await prefs.setString(_categoriesKey, jsonEncode(cats));
      await _syncTagsUnlocked(null, cats);
    });
  }

  // ── Private helper — non-verrouillé ──────────────────────────────────────
  // Appelé UNIQUEMENT depuis l'intérieur d'un bloc _lock.synchronized.
  // NE PAS appeler _lock.synchronized() ici (Lock non-réentrant).
  // T-3-01: brand values already sanitized before reaching here.
  // Tag sync failure is non-fatal — local SharedPreferences write is source of truth.
  static Future<void> _syncTagsUnlocked(
      List<String>? brands, List<String>? categories) async {
    try {
      if (brands != null) {
        await OneSignal.User.addTagWithKey(
          'subscribed_brands',
          brands.join(','),
        );
      }
      if (categories != null) {
        await OneSignal.User.addTagWithKey(
          'subscribed_categories',
          categories.join(','),
        );
      }
    } catch (_) {
      // OneSignal tag sync failure is non-fatal — subscriptions persist locally
    }
  }
}
```

// test/subscription_service_test.dart
// Unit tests for ALRT-01 (brands) and ALRT-02 (categories).
// Pattern: mirrors test/read_status_service_test.dart
import 'package:flutter_test/flutter_test.dart';
import 'package:shared_preferences/shared_preferences.dart';
import 'package:vigiconso/services/subscription_service.dart';

void main() {
  setUp(() {
    SharedPreferences.setMockInitialValues({});
  });

  group('SubscriptionService — brands (ALRT-01)', () {
    test('addBrand persists brand to SharedPreferences', () async {
      await SubscriptionService.addBrand('Nestlé');
      final brands = await SubscriptionService.getBrands();
      expect(brands, contains('Nestlé'));
    });

    test('addBrand is idempotent — duplicate adds do not create duplicate entries', () async {
      await SubscriptionService.addBrand('Nestlé');
      await SubscriptionService.addBrand('Nestlé');
      final brands = await SubscriptionService.getBrands();
      expect(brands.where((b) => b.toLowerCase() == 'nestlé').length, equals(1));
    });

    test('removeBrand removes the brand', () async {
      await SubscriptionService.addBrand('Nestlé');
      await SubscriptionService.removeBrand('Nestlé');
      final brands = await SubscriptionService.getBrands();
      expect(brands, isNot(contains('Nestlé')));
    });

    test('getBrands returns empty list when no brands stored', () async {
      final brands = await SubscriptionService.getBrands();
      expect(brands, isEmpty);
    });

    test('addBrand trims whitespace', () async {
      await SubscriptionService.addBrand('  Samsung  ');
      final brands = await SubscriptionService.getBrands();
      expect(brands, contains('Samsung'));
      expect(brands, isNot(contains('  Samsung  ')));
    });

    test('addBrand ignores empty brand after trim', () async {
      await SubscriptionService.addBrand('   ');
      final brands = await SubscriptionService.getBrands();
      expect(brands, isEmpty);
    });
  });

  group('SubscriptionService — categories (ALRT-02)', () {
    test('addCategory persists category key', () async {
      await SubscriptionService.addCategory('alimentation');

    test('removeCategory removes the category', () async {
      await SubscriptionService.addCategory('alimentation');
      await SubscriptionService.removeCategory('alimentation');
      final cats = await SubscriptionService.getCategories();
      expect(cats, isNot(contains('alimentation')));
    });

    test('getCategories returns empty list when nothing stored', () async {
      final cats = await SubscriptionService.getCategories();
      expect(cats, isEmpty);
    });

    test('brands and categories are stored independently', () async {
      await SubscriptionService.addBrand('Nestlé');
      await SubscriptionService.addCategory('alimentation');

      final cats = await SubscriptionService.getCategories();

      expect(cats, contains('alimentation'));
      expect(brands, isNot(contains('alimentation')));

  });
