**Step 1 — Create lib/providers/mes_alertes_provider.dart:**

Derived synchronous `Provider<AsyncValue<List<Rappel>>>`. Watches both rappelServiceProvider and subscriptionProvider. Propagates loading/error state from either upstream. Returns empty list (not all rappels) when subscriptions.isEmpty.

```dart
// lib/providers/mes_alertes_provider.dart
// Derived Provider — no API call. Pure in-memory filter of rappelServiceProvider.
// Pattern: synchronous Provider<AsyncValue<T>> watching two async providers.
// [PITFALL]: When subscriptions.isEmpty, return empty list (not all rappels).
// [PITFALL]: Use r.marqueProduit (typed) NOT r['marque_produit'] (dynamic map access).

import 'package:flutter_riverpod/flutter_riverpod.dart';
import 'package:vigiconso/models/rappel.dart';
import 'package:vigiconso/models/subscriptions.dart';
import 'package:vigiconso/providers/rappel_provider.dart';
import 'package:vigiconso/providers/subscription_provider.dart';

/// Returns AsyncValue<List<Rappel>> filtered by user subscriptions.
/// Propagates loading/error state from either upstream provider.
/// Returns AsyncValue.data([]) (empty) when subscriptions are empty — never returns all rappels.
final mesAlertesProvider = Provider<AsyncValue<List<Rappel>>>((ref) {
  final rappelsAsync = ref.watch(rappelServiceProvider);
  final subsAsync = ref.watch(subscriptionProvider);

  // Propagate loading state from either upstream
  if (rappelsAsync is AsyncLoading || subsAsync is AsyncLoading) {
    return const AsyncValue.loading();
  }
  // Propagate error state
  if (rappelsAsync is AsyncError) return rappelsAsync;
  if (subsAsync is AsyncError) return subsAsync;

  final rappels = rappelsAsync.value ?? [];
  final subs = subsAsync.value ?? const Subscriptions(brands: [], categories: []);

  // Guard: no subscriptions → empty feed (not all rappels)
  if (subs.isEmpty) {
    return const AsyncValue.data([]);
  }

  final filtered = rappels.where((r) {
    // Brand match — case-insensitive contains
    // r.marqueProduit is typed String (from Rappel model)
    final marque = r.marqueProduit.toLowerCase();
    final brandMatch = subs.brands.any(
      (b) => marque.contains(b.toLowerCase()),
    );

    // Category match — use contains on lowercase key against lowercase label
    // categorieProduit may be 'Alimentation > Fruits et légumes' — key 'alimentation' matches via contains
    final categorie = (r.categorieProduit ?? '').toLowerCase();
    final catMatch = subs.categories.any(
      (c) => categorie.contains(c.toLowerCase()),
    );

    return brandMatch || catMatch;
  }).toList();

  return AsyncValue.data(filtered);
});
```

```dart
// lib/screens/mes_alertes_screen.dart
// Feed screen for ALRT-05 — displays filtered rappels from mesAlertesProvider.
// Pattern: mirrors lib/screens/rappel_screen.dart (ConsumerWidget + AsyncValue.when).
// IMPORTANT: Pass rappel.toJson() to RappelDetailsPage — page still uses dynamic map access.

import 'package:flutter/material.dart';
import 'package:flutter_riverpod/flutter_riverpod.dart';
import 'package:vigiconso/models/rappel.dart';
import 'package:vigiconso/providers/mes_alertes_provider.dart';
import 'package:vigiconso/screens/rappel_details_page.dart';
import 'package:vigiconso/widgets/shimmer_loading.dart';

class MesAlertesScreen extends ConsumerWidget {
  const MesAlertesScreen({super.key});

  void _openDetails(BuildContext context, Rappel rappel) {
    Navigator.push(
      context,
      MaterialPageRoute(
        builder: (_) => RappelDetailsPage(rappel: rappel.toJson()),
      ),
    );
  }

  Widget _buildEmptyState(BuildContext context) {
    final cs = Theme.of(context).colorScheme;
    return Center(
      child: Padding(
        padding: const EdgeInsets.all(32),
        child: Column(
          mainAxisAlignment: MainAxisAlignment.center,
          children: [
            Container(
              padding: const EdgeInsets.all(20),
              decoration: BoxDecoration(
                color: cs.primaryContainer,
                shape: BoxShape.circle,
              ),
              child: Icon(
                Icons.notifications_none_outlined,
                size: 48,
                color: cs.onPrimaryContainer,
              ),
            ),
            const SizedBox(height: 16),
            Text(
              'Aucun rappel pour vos abonnements',
              style: Theme.of(context).textTheme.titleMedium,
              textAlign: TextAlign.center,
            ),
            const SizedBox(height: 8),
            Text(
              'Ajoutez des marques ou catégories depuis l\'onglet Abonnements.',
              style: Theme.of(context)
                  .textTheme
                  .bodyMedium
                  ?.copyWith(color: cs.onSurfaceVariant),
              textAlign: TextAlign.center,
            ),
          ],
        ),
      ),
    );
  }

  @override
  Widget build(BuildContext context, WidgetRef ref) {
    final alertesAsync = ref.watch(mesAlertesProvider);
    return alertesAsync.when(
      loading: () => ListView.builder(
        padding: const EdgeInsets.symmetric(vertical: 8),
        itemCount: 5,
        itemBuilder: (_, __) => const RappelCardShimmer(),
      ),
      error: (e, _) => Center(child: Text('Erreur : $e')),
      data: (rappels) => rappels.isEmpty
          ? _buildEmptyState(context)
          : ListView.builder(
              padding: const EdgeInsets.symmetric(vertical: 8),
              itemCount: rappels.length,
              itemBuilder: (ctx, i) => ListTile(
                title: Text(rappels[i].libelle),
                subtitle: Text(rappels[i].marqueProduit),
                onTap: () => _openDetails(ctx, rappels[i]),
              ),
            ),
    );
  }
}
```

```dart
// ALRT-04 CRUD management screen + ALRT-05 feed via inner DefaultTabController.
// Pattern: mirrors lib/screens/favorites_screen.dart (ConsumerStatefulWidget + AsyncValue.when).
// Two sub-tabs: "Fil" (MesAlertesScreen inline) + "Abonnements" (brands + categories CRUD).
// Category subscription uses FilterChip — toggles on tap via subscriptionProvider.notifier.

import 'package:flutter/material.dart';
import 'package:flutter_riverpod/flutter_riverpod.dart';
import 'package:vigiconso/providers/subscription_provider.dart';
import 'package:vigiconso/screens/mes_alertes_screen.dart';
import 'package:vigiconso/utils/constants.dart';
import 'package:vigiconso/widgets/shimmer_loading.dart';

class AlertesScreen extends ConsumerStatefulWidget {
  const AlertesScreen({super.key});

  @override
  ConsumerState<AlertesScreen> createState() => _AlertesScreenState();
}

class _AlertesScreenState extends ConsumerState<AlertesScreen> {
  final _addBrandController = TextEditingController();

  @override
  void dispose() {
    _addBrandController.dispose();
    super.dispose();
  }

  void _showAddBrandDialog() {
    _addBrandController.clear();
    showDialog(
      context: context,
      builder: (_) => AlertDialog(
        title: const Text('Suivre une marque'),
        content: TextField(
          controller: _addBrandController,
          autofocus: true,
          decoration: const InputDecoration(
            hintText: 'Ex: Nestlé, Samsung...',
            border: OutlineInputBorder(),
          ),
          textCapitalization: TextCapitalization.words,
        ),
        actions: [
          TextButton(
            onPressed: () => Navigator.pop(context),
            child: const Text('Annuler'),
          ),
          FilledButton(
            onPressed: () {
              final brand = _addBrandController.text.trim();
              if (brand.isNotEmpty) {
                ref.read(subscriptionProvider.notifier).addBrand(brand);
              }
              Navigator.pop(context);
            },
            child: const Text('Suivre'),
          ),
        ],
      ),
    );
  }

  Widget _buildAbonnementsTab(BuildContext context) {
    final subsAsync = ref.watch(subscriptionProvider);
    return subsAsync.when(
      loading: () => ListView.builder(
        itemCount: 4,
        itemBuilder: (_, __) => const RappelCardShimmer(),
      ),
      error: (e, _) => Center(child: Text('Erreur : $e')),
      data: (subs) => ListView(
        padding: const EdgeInsets.all(16),
        children: [
          // ── Marques suivies ─────────────────────────────────────────────
          Row(
            mainAxisAlignment: MainAxisAlignment.spaceBetween,
            children: [
              Text(
                'Marques suivies',
                style: Theme.of(context).textTheme.titleMedium,
              ),
              IconButton(
                icon: const Icon(Icons.add),
                tooltip: 'Ajouter une marque',
                onPressed: _showAddBrandDialog,
              ),
            ],
          ),
          if (subs.brands.isEmpty)
            Padding(
              padding: const EdgeInsets.symmetric(vertical: 8),
              child: Text(
                'Aucune marque suivie.',
                style: Theme.of(context).textTheme.bodyMedium?.copyWith(
                      color: Theme.of(context).colorScheme.onSurfaceVariant,
                    ),
              ),
            )
          else
            ...subs.brands.map(
              (brand) => ListTile(
                title: Text(brand),
                trailing: IconButton(
                  icon: const Icon(Icons.delete_outline),
                  tooltip: 'Ne plus suivre',
                  onPressed: () {
                    ref.read(subscriptionProvider.notifier).removeBrand(brand);
                    ScaffoldMessenger.of(context).showSnackBar(
                      SnackBar(
                        content: Text('Marque "$brand" retirée'),
                        duration: const Duration(seconds: 2),
                        behavior: SnackBarBehavior.floating,
                      ),
                    );
                  },
                ),
              ),
```
